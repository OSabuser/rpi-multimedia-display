# Отключить встроенный звук (используем hifiberry-dac)
dtparam=audio=off

# Adding support of the custom TFT display
# RGB 10.1" 60 Hz fps WxHx60Hz = 37 MHz
hdmi_timings=1024 0 70 90 75 600 0 1 23 0 0 0 0 60 0 37000000 1 
hdmi_group=2
hdmi_mode=87
hdmi_force_hotplug=1
config_hdmi_boost=7
gpu_mem=128
hdmi_drive=1
display_hdmi_rotate=3

[all]
# FOR RGB DISPLAY
# Enable LCD Driver
gpio=12=op,dh 
# Enable LCD Backlight
gpio=16=op,dh
# LCD L/R
gpio=3=op,dh
# LCD U/D
gpio=4=op,dl

# BUTTON PINS
gpio=7=ip,pu
gpio=8=ip,pu

# ENABLE UART
dtoverlay=disable-bt
dtparam=act_led_trigger=actpwr
enable_uart=1

# ENABLE MAX98357 I2S SUPPORT
dtoverlay=hifiberry-dac
dtoverlay=i2s-mmap
